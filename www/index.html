let wakeLock = null; async function bloquerMiseEnVeille() { try { if ('wakeLock' in navigator) { wakeLock = await navigator.wakeLock.request('screen'); console.log("Écran verrouillé (allumé)."); } } catch (e) { console.error("Erreur WakeLock:", e); } } async function autoriserMiseEnVeille() { try { if (wakeLock !== null) { await wakeLock.release(); wakeLock = null; console.log("Mise en veille autorisée."); } } catch (e) { console.error("Erreur WakeLock Release:", e); } }
